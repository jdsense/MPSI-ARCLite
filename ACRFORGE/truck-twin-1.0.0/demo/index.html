트럭 트윈 데모

YARD-OPS 배차실행 화면의 3D 트럭을 바닐라 three.js 단일 파일로 옮긴 그리기 전용 라이브러리.

<script src="truck-twin.min.js?key=발급받은_API_키"></script>
<div id="stage" style="height: 520px"></div>
<script>
  const twin = TruckTwin.create(document.getElementById("stage"), {
    truck: { label, vehicleTypeName, bedLengthMm, heightLimitMm, usableHeightMm, bedFloorHeightMm, topSupportMm },
    cargo: [{ key, compartment, stackOrder, label, blockLabel, lengthMm, loadHeightMm, packingHeightMm, colorIndex, detail }],
    theme: "auto",                       // 'light' | 'dark' | 'auto'
    onSelect: (key) => {},             // 3D 에서 화물을 클릭하면
    onHover: (key) => {},
    onFail: (message, reason) => {},   // 키 오류 · WebGL 불가 · 컨텍스트 손실
  });
  twin.select("P-001");   // 바깥(2D 목록)에서 강조 맞추기
  twin.setData(truck, cargo);
</script>
